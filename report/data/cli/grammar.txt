  0. program -> decls

  9. plist -> IDENT

 15. stmt -> PRINT expr SEMI
 16. stmt -> RETURN expr SEMI
 17. stmt -> IF expr THEN stmts END
 18. stmt -> IF expr THEN stmts ELSE stmts END

 26. expr -> join

 30. eq -> eq NE rel
 31. eq -> rel

 35. rel -> rel GE add

 38. add -> add MINUS term

 41. term -> term SLASH unary
 42. term -> term PERCENT unary
 43. term -> unary
 44. unary -> MINUS unary
 45. unary -> NOT unary

 48. factor -> STRING

 50. factor -> FALSE
